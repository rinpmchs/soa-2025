---- register new user ----

curl -X POST http://localhost:8001/api/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "tatatata", "email": "tatatata@example.com", "password": "tatatata"}'


---- not enough symbols in password ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "tata", "email": "aaaaa@example.com", "password": "tata"}'


---- user already exists ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "tatatata", "email": "arinash@example.com", "password": "tatatata"}'


---- login or password incorrect ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "email": "arinash@example.com", "password": "arina123"}'


---- get token ----
curl -X POST http://localhost:8001/api/users/token \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "password": "arina111"}'

curl -X POST http://localhost:8001/api/users/token \
  -H "Content-Type: application/json" \
  -d '{"login": "tatatata", "password": "tatatata"}'

curl -X POST http://localhost:8001/api/users/token \
  -H "Content-Type: application/json" \
  -d '{"login": "teach", "password": "TAstr0ngPASS"}'


---- get user ----
curl -X GET http://localhost:8001/api/users/me \
  -H "Authorization: Bearer ACCESS_TOKEN"   (тут мы просто вставляем сгенерированный при логине токен)


---- update user ----
curl -X PUT http://localhost:8001/api/users/me \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer ACCESS_TOKEN" \
  -d '{"email": "new-email@example.com"}'
