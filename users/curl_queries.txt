---- register new user ----

curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "email": "arinash@example.com", "password": "arina111"}'


---- not enough symbols in password ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "aaaaa", "email": "aaaaa@example.com", "password": "arina"}'


---- user already exists ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "email": "arinash@example.com", "password": "arina111"}'


---- login or password incorrect ----
curl -X POST http://localhost:8000/users/register \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "email": "arinash@example.com", "password": "arina123"}'


---- login ----
curl -X POST http://localhost:8000/users/login_json \
  -H "Content-Type: application/json" \
  -d '{"login": "arinasher", "password": "arina111"}'

curl -X POST http://localhost:8000/users/login_json \
  -H "Content-Type: application/json" \
  -d '{"login": "arinash", "password": "arinapass"}'

curl -X POST http://localhost:8000/users/login_json \
  -H "Content-Type: application/json" \
  -d '{"login": "teach", "password": "TAstr0ngPASS"}'


---- get user ----
curl -X GET http://localhost:8000/users/me \
  -H "Authorization: Bearer ACCESS_TOKEN"   (тут мы просто вставляем сгенерированный при логине токен)


---- update user ----
curl -X PUT http://localhost:8000/users/update \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer ACCESS_TOKEN" \
  -d '{"email": "new-email@example.com"}'
